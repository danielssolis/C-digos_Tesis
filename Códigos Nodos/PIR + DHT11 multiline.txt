// main.c - NODO HIJO PIR + DHT11 ALINEADO CON ROOT NUEVO
// ESP-IDF v6.1
//
// Compatible con root actual:
// struct msg_t:
//   uint8_t value_type;   // 0=int, 1=float
//   char node_name[24];
//   char zone[24];
//   char sensor[24];
//   char metric[16];
//   char unit[16];
//   union {
//       int32_t i;
//       float f;
//   } v;
//   uint32_t timestamp;
//
// Lógica conservada:
// - PIR: envío inmediato cuando cambia + envío periódico aunque no cambie
// - DHT11: envío periódico de temperatura y humedad

static const char *TAG = "Cocina_PIR_DHT11";

/* =========================
 * IDENTIDAD DEL NODO
 * ========================= */
#define NODE_NAME               "Cocina_nodo_1"
#define NODE_ZONE               "Cocina_PIR_DHT11"

/* PIR */
#define PIR_SENSOR_NAME         "Pir_cocina"
#define PIR_METRIC_NAME         "movement"
#define PIR_UNIT_NAME           "bool"

/* DHT11 */
#define DHT_SENSOR_NAME         "Dht11_cocina"
#define DHT_METRIC_TEMP         "temperature"
#define DHT_UNIT_TEMP           "C"
#define DHT_METRIC_HUM          "humidity"
#define DHT_UNIT_HUM            "pct"

/* =========================
 * MESH CONFIG
 * ========================= */
#define MESH_ID        {0x32,0x10,0x88,0x44,0x55,0x66}
#define ROUTER_SSID    "Pdegrado24"
#define ROUTER_PASS    "20Pgrado26$"
#define ROUTER_CH      0
#define MESH_AP_PASS   "v2345678"
#define MESH_MAX_CONN  6

/* =========================
 * PROTOCOLO ALINEADO CON ROOT
 * ========================= */
typedef enum {
    VALUE_TYPE_INT   = 0,
    VALUE_TYPE_FLOAT = 1
} value_type_t;

typedef struct __attribute__((packed)) {
    uint8_t value_type;   // 0=int, 1=float
    char    node_name[24];
    char    zone[24];
    char    sensor[24];
    char    metric[16];
    char    unit[16];
    union {
        int32_t i;
        float   f;
    } v;
    uint32_t timestamp;
} msg_t;

/* =========================
 * PIR CONFIG
 * ========================= */
#define PIR_GPIO                26
#define PIR_SEND_PERIOD_MS      5000

/* =========================
 * DHT11 CONFIG
 * ========================= */
#define DHT11_GPIO              27
#define DHT11_SEND_PERIOD_MS    10000
#define DHT11_MIN_INTERVAL_MS   2000

/* =========================
 * STARTUP JITTER
 * ========================= */
#define STARTUP_JITTER_MIN_MS   1000
#define STARTUP_JITTER_MAX_MS   5000

/* =====================================================
 * HELPERS
 * ===================================================== */
static void safe_copy_fixed(char *dst, size_t dst_sz, const char *src)
{
    if (!dst || dst_sz == 0) return;

    memset(dst, 0, dst_sz);

    if (!src) return;

    snprintf(dst, dst_sz, "%s", src);
}

static uint32_t get_timestamp_s(void)
{
    time_t now = time(NULL);

    if ((uint32_t)now >= 1000000000UL) {
        return (uint32_t)now;
    }

    return (uint32_t)(esp_timer_get_time() / 1000000ULL);
}

static void apply_startup_jitter(void)
{
    uint32_t span = STARTUP_JITTER_MAX_MS - STARTUP_JITTER_MIN_MS;
    uint32_t jitter = STARTUP_JITTER_MIN_MS;

    if (span > 0) {
        jitter += (esp_random() % (span + 1));
    }

    ESP_LOGI(TAG, "Aplicando startup jitter de %lu ms", (unsigned long)jitter);
    vTaskDelay(pdMS_TO_TICKS(jitter));
}

/* =====================================================
 * EVENTOS WIFI / IP / MESH
 * ===================================================== */
static void wifi_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != WIFI_EVENT) return;

    switch (id) {
        case WIFI_EVENT_STA_START:
            ESP_LOGI(TAG, "WIFI_EVENT_STA_START");
            break;

        case WIFI_EVENT_STA_CONNECTED:
            ESP_LOGI(TAG, "WIFI_EVENT_STA_CONNECTED");
            break;

        case WIFI_EVENT_STA_DISCONNECTED:
            ESP_LOGW(TAG, "WIFI desconectado");
            break;

        default:
            break;
    }
}

static void ip_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != IP_EVENT || id != IP_EVENT_STA_GOT_IP) return;

    ip_event_got_ip_t *event = (ip_event_got_ip_t *)data;
    ESP_LOGI(TAG, "IP obtenida " IPSTR, IP2STR(&event->ip_info.ip));
}

static void mesh_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != MESH_EVENT) return;

    switch (id) {
        case MESH_EVENT_STARTED: {
            mesh_addr_t myid = {0};
            esp_mesh_get_id(&myid);
            ESP_LOGI(TAG, "Mesh iniciado ID=%02X:%02X:%02X:%02X:%02X:%02X",
                     myid.addr[0], myid.addr[1], myid.addr[2],
                     myid.addr[3], myid.addr[4], myid.addr[5]);
            break;
        }

        case MESH_EVENT_PARENT_CONNECTED:
            ESP_LOGI(TAG, "Parent conectado. layer=%d root=%d",
                     esp_mesh_get_layer(), esp_mesh_is_root());
            break;

        case MESH_EVENT_PARENT_DISCONNECTED:
            ESP_LOGW(TAG, "Parent desconectado. layer=%d",
                     esp_mesh_get_layer());
            break;

        default:
            break;
    }
}

/* =====================================================
 * ENVÍO MESH
 * ===================================================== */
static esp_err_t mesh_send_msg(msg_t *msg)
{
    mesh_data_t data = {
        .data  = (uint8_t *)msg,
        .size  = sizeof(msg_t),
        .proto = MESH_PROTO_BIN,
        .tos   = MESH_TOS_P2P
    };

    return esp_mesh_send(NULL, &data, 0, NULL, 0);
}

static int mesh_send_int_metric(const char *sensor, const char *metric, const char *unit, int32_t value)
{
    if (esp_mesh_get_layer() <= 0) {
        ESP_LOGW(TAG, "Sin layer mesh para enviar int");
        return -1;
    }

    msg_t msg;
    memset(&msg, 0, sizeof(msg));

    msg.value_type = VALUE_TYPE_INT;

    safe_copy_fixed(msg.node_name, sizeof(msg.node_name), NODE_NAME);
    safe_copy_fixed(msg.zone, sizeof(msg.zone), NODE_ZONE);
    safe_copy_fixed(msg.sensor, sizeof(msg.sensor), sensor);
    safe_copy_fixed(msg.metric, sizeof(msg.metric), metric);
    safe_copy_fixed(msg.unit, sizeof(msg.unit), unit);

    msg.v.i = value;
    msg.timestamp = get_timestamp_s();

    esp_err_t err = mesh_send_msg(&msg);
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "mesh_send int FAIL err=%s", esp_err_to_name(err));
        return -1;
    }

    ESP_LOGI(TAG, "TX INT sensor=%s metric=%s unit=%s val=%" PRId32 " ts=%" PRIu32,
             sensor, metric, unit, value, msg.timestamp);
    return 0;
}

static int mesh_send_float_metric(const char *sensor, const char *metric, const char *unit, float value)
{
    if (esp_mesh_get_layer() <= 0) {
        ESP_LOGW(TAG, "Sin layer mesh para enviar float");
        return -1;
    }

    msg_t msg;
    memset(&msg, 0, sizeof(msg));

    msg.value_type = VALUE_TYPE_FLOAT;

    safe_copy_fixed(msg.node_name, sizeof(msg.node_name), NODE_NAME);
    safe_copy_fixed(msg.zone, sizeof(msg.zone), NODE_ZONE);
    safe_copy_fixed(msg.sensor, sizeof(msg.sensor), sensor);
    safe_copy_fixed(msg.metric, sizeof(msg.metric), metric);
    safe_copy_fixed(msg.unit, sizeof(msg.unit), unit);

    msg.v.f = value;
    msg.timestamp = get_timestamp_s();

    esp_err_t err = mesh_send_msg(&msg);
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "mesh_send float FAIL err=%s", esp_err_to_name(err));
        return -1;
    }

    ESP_LOGI(TAG, "TX FLOAT sensor=%s metric=%s unit=%s val=%.2f ts=%" PRIu32,
             sensor, metric, unit, value, msg.timestamp);
    return 0;
}

/* =====================================================
 * DHT11 LOW LEVEL
 * ===================================================== */
static void dht11_set_output(void)
{
    gpio_set_direction(DHT11_GPIO, GPIO_MODE_OUTPUT);
}

static void dht11_set_input(void)
{
    gpio_set_direction(DHT11_GPIO, GPIO_MODE_INPUT);
}

static int dht11_wait_for_level(int level, uint32_t timeout_us)
{
    uint32_t start = (uint32_t)esp_timer_get_time();

    while (gpio_get_level(DHT11_GPIO) == level) {
        if (((uint32_t)esp_timer_get_time() - start) > timeout_us) {
            return -1;
        }
    }
    return 0;
}

static esp_err_t dht11_read(float *temperature, float *humidity)
{
    if (!temperature || !humidity) {
        return ESP_ERR_INVALID_ARG;
    }

    uint8_t data[5] = {0};

    /* Señal de inicio */
    dht11_set_output();
    gpio_set_level(DHT11_GPIO, 0);
    vTaskDelay(pdMS_TO_TICKS(20));   // >=18 ms
    gpio_set_level(DHT11_GPIO, 1);
    ets_delay_us(30);
    dht11_set_input();

    /* Respuesta del sensor */
    if (dht11_wait_for_level(1, 100) != 0) return ESP_ERR_TIMEOUT;
    if (dht11_wait_for_level(0, 100) != 0) return ESP_ERR_TIMEOUT;
    if (dht11_wait_for_level(1, 100) != 0) return ESP_ERR_TIMEOUT;

    /* Leer 40 bits */
    for (int i = 0; i < 40; i++) {
        if (dht11_wait_for_level(0, 70) != 0) return ESP_ERR_TIMEOUT;

        uint32_t start = (uint32_t)esp_timer_get_time();

        if (dht11_wait_for_level(1, 100) != 0) return ESP_ERR_TIMEOUT;

        uint32_t pulse_len = (uint32_t)esp_timer_get_time() - start;

        data[i / 8] <<= 1;
        if (pulse_len > 40) {
            data[i / 8] |= 1;
        }
    }

    uint8_t checksum = (uint8_t)(data[0] + data[1] + data[2] + data[3]);
    if (checksum != data[4]) {
        return ESP_ERR_INVALID_CRC;
    }

    /* DHT11: normalmente parte decimal = 0 */
    *humidity = (float)data[0];
    *temperature = (float)data[2];

    return ESP_OK;
}

/* =====================================================
 * TASK PIR
 * ===================================================== */
static void pir_task(void *arg)
{
    int last_pir_value = gpio_get_level(PIR_GPIO);
    bool first = true;

    while (esp_mesh_get_layer() <= 0) {
        vTaskDelay(pdMS_TO_TICKS(200));
    }

    apply_startup_jitter();

    ESP_LOGI(TAG, "Nodo PIR listo. layer=%d root=%d PIR GPIO=%d",
             esp_mesh_get_layer(), esp_mesh_is_root(), PIR_GPIO);

    while (1) {
        if (esp_mesh_get_layer() <= 0) {
            ESP_LOGW(TAG, "PIR sin layer. Esperando reconexión...");
            while (esp_mesh_get_layer() <= 0) {
                vTaskDelay(pdMS_TO_TICKS(500));
            }
            ESP_LOGI(TAG, "PIR reconectado");
            last_pir_value = gpio_get_level(PIR_GPIO);
            first = true;
        }

        int pir_value = gpio_get_level(PIR_GPIO);

        if (pir_value != last_pir_value || first) {
            if (mesh_send_int_metric(PIR_SENSOR_NAME, PIR_METRIC_NAME, PIR_UNIT_NAME, pir_value) == 0) {
                ESP_LOGI(TAG, "Enviado PIR (cambio): %d", pir_value);
            }
            last_pir_value = pir_value;
            first = false;
        }

        if (mesh_send_int_metric(PIR_SENSOR_NAME, PIR_METRIC_NAME, PIR_UNIT_NAME, pir_value) == 0) {
            ESP_LOGI(TAG, "Enviado PIR (periodico): %d (cada %d ms)",
                     pir_value, PIR_SEND_PERIOD_MS);
        }

        vTaskDelay(pdMS_TO_TICKS(PIR_SEND_PERIOD_MS));
    }
}

/* =====================================================
 * TASK DHT11
 * ===================================================== */
static void dht11_task(void *arg)
{
    uint32_t last_read_ms = 0;

    while (esp_mesh_get_layer() <= 0) {
        vTaskDelay(pdMS_TO_TICKS(200));
    }

    vTaskDelay(pdMS_TO_TICKS(1500));

    ESP_LOGI(TAG, "Nodo DHT11 listo. GPIO=%d", DHT11_GPIO);

    while (1) {
        if (esp_mesh_get_layer() <= 0) {
            ESP_LOGW(TAG, "DHT11 sin layer. Esperando reconexión...");
            while (esp_mesh_get_layer() <= 0) {
                vTaskDelay(pdMS_TO_TICKS(500));
            }
            ESP_LOGI(TAG, "DHT11 reconectado");
        }

        uint32_t now_ms = (uint32_t)(esp_timer_get_time() / 1000ULL);
        if ((now_ms - last_read_ms) < DHT11_MIN_INTERVAL_MS) {
            vTaskDelay(pdMS_TO_TICKS(DHT11_MIN_INTERVAL_MS - (now_ms - last_read_ms)));
        }

        float temp = 0.0f;
        float hum  = 0.0f;

        esp_err_t err = dht11_read(&temp, &hum);
        last_read_ms = (uint32_t)(esp_timer_get_time() / 1000ULL);

        if (err == ESP_OK) {
            ESP_LOGI(TAG, "DHT11 OK temp=%.1fC hum=%.1f%%", temp, hum);

            mesh_send_float_metric(DHT_SENSOR_NAME, DHT_METRIC_TEMP, DHT_UNIT_TEMP, temp);
            vTaskDelay(pdMS_TO_TICKS(100));

            mesh_send_float_metric(DHT_SENSOR_NAME, DHT_METRIC_HUM, DHT_UNIT_HUM, hum);
        } else {
            ESP_LOGW(TAG, "DHT11 read FAIL: %s", esp_err_to_name(err));
        }

        vTaskDelay(pdMS_TO_TICKS(DHT11_SEND_PERIOD_MS));
    }
}

/* =====================================================
 * APP MAIN
 * ===================================================== */
void app_main(void)
{
    esp_err_t err = nvs_flash_init();
    if (err == ESP_ERR_NVS_NO_FREE_PAGES ||
        err == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        err = nvs_flash_init();
    }
    ESP_ERROR_CHECK(err);

    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());

    ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(IP_EVENT, IP_EVENT_STA_GOT_IP, &ip_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(MESH_EVENT, ESP_EVENT_ANY_ID, &mesh_event_handler, NULL));

    esp_netif_t *netif_sta = NULL;
    esp_netif_t *netif_ap  = NULL;
    ESP_ERROR_CHECK(esp_netif_create_default_wifi_mesh_netifs(&netif_sta, &netif_ap));

    wifi_init_config_t wifi_cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&wifi_cfg));
    ESP_ERROR_CHECK(esp_wifi_set_storage(WIFI_STORAGE_FLASH));
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_APSTA));
    ESP_ERROR_CHECK(esp_wifi_set_ps(WIFI_PS_NONE));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_ERROR_CHECK(esp_mesh_init());

    mesh_cfg_t mc = MESH_INIT_CONFIG_DEFAULT();
    uint8_t mid[6] = MESH_ID;
    memcpy(mc.mesh_id.addr, mid, 6);

    mc.channel = ROUTER_CH;

    mc.router.ssid_len = strlen(ROUTER_SSID);
    memcpy(mc.router.ssid, ROUTER_SSID, mc.router.ssid_len);
    memcpy(mc.router.password, ROUTER_PASS, strlen(ROUTER_PASS));

    mc.mesh_ap.max_connection = MESH_MAX_CONN;
    mc.mesh_ap.nonmesh_max_connection = 0;
    memcpy(mc.mesh_ap.password, MESH_AP_PASS, strlen(MESH_AP_PASS));

    ESP_ERROR_CHECK(esp_mesh_set_config(&mc));
    ESP_ERROR_CHECK(esp_mesh_set_self_organized(true, false));
    ESP_ERROR_CHECK(esp_mesh_start());

    ESP_LOGI(TAG, "Nodo PIR + DHT11 iniciado. NO ROOT");
    ESP_LOGI(TAG, "sizeof(msg_t)=%u", (unsigned)sizeof(msg_t));
    ESP_LOGI(TAG, "PIR_SEND_PERIOD_MS=%d", PIR_SEND_PERIOD_MS);
    ESP_LOGI(TAG, "DHT11_SEND_PERIOD_MS=%d", DHT11_SEND_PERIOD_MS);

    gpio_config_t pir_conf = {
        .pin_bit_mask = (1ULL << PIR_GPIO),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    ESP_ERROR_CHECK(gpio_config(&pir_conf));

    gpio_config_t dht_conf = {
        .pin_bit_mask = (1ULL << DHT11_GPIO),
        .mode = GPIO_MODE_INPUT_OUTPUT_OD,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    ESP_ERROR_CHECK(gpio_config(&dht_conf));
    gpio_set_level(DHT11_GPIO, 1);

    xTaskCreate(pir_task, "pir_task", 4096, NULL, 5, NULL);
    xTaskCreate(dht11_task, "dht11_task", 4096, NULL, 5, NULL);
}
