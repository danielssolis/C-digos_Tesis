/* =========================
 * TAG
 * ========================= */
static const char *TAG = "HCSR04_sala3";

/* =========================
 * IDENTIDAD DEL NODO
 * ========================= */
#define NODE_NAME        "Sala_nodo_3"
#define NODE_ZONE        "Sala"
#define NODE_TYPE        "Distancia"
#define SENSOR_NAME      "Proxi_sofa"

/* =========================
 * MESH CONFIG
 * ========================= */
#define MESH_ID        {0x32,0x10,0x88,0x44,0x55,0x66}

#define ROUTER_SSID    "Pdegrado24"
#define ROUTER_PASS    "20Pgrado26$"
#define ROUTER_CH      0
#define MESH_AP_PASS   "v2345678"
#define MESH_MAX_CONN  6

/* =========================
 * PROTOCOLO V2
 * ========================= */
typedef enum {
    SENSOR_PIR        = 1,
    SENSOR_TEMP       = 2,
    SENSOR_HUM        = 3,
    SENSOR_DOOR       = 4,
    SENSOR_BUTTON     = 5,
    SENSOR_LIGHT      = 6,
    SENSOR_CURRENT    = 7,
    SENSOR_FLOW       = 8,
    SENSOR_DISTANCE   = 9,
    SENSOR_GAS_MQ4    = 10,
    SENSOR_VIBRATION  = 11,
    SENSOR_C1001      = 12,
    SENSOR_OTHER      = 255
} sensor_type_t;

typedef enum {
    VALUE_INT   = 1,
    VALUE_FLOAT = 2
} value_type_t;

typedef struct __attribute__((packed)) {
    uint8_t  sensor_type;
    uint8_t  value_type;
    uint16_t reserved;

    char     node_name[24];
    char     zone[24];
    char     node_type[24];
    char     sensor_name[24];
    char     metric_name[24];
    char     unit[12];

    union {
        int32_t i;
        float   f;
    } v;

    uint32_t timestamp;
} sensor_msg_v2_t;

/* =========================
 * HC-SR04 CONFIG
 * =========================
 * TRIG: salida 3.3V
 * ECHO: salida 5V -> DEBES BAJAR A 3.3V antes del ESP32
 */
#define HCSR04_TRIG_GPIO       27
#define HCSR04_ECHO_GPIO       26

#define SEND_PERIOD_MS         10000
#define STARTUP_JITTER_MIN_MS  1000
#define STARTUP_JITTER_MAX_MS  5000

#define ECHO_WAIT_TIMEOUT_US   30000
#define MAX_DISTANCE_CM        400.0f

/* =====================================================
 * HELPERS GENERALES
 * ===================================================== */
static void safe_copy_fixed(char *dst, size_t dst_sz, const char *src)
{
    if (!dst || dst_sz == 0) {
        return;
    }

    memset(dst, 0, dst_sz);

    if (!src) {
        return;
    }

    snprintf(dst, dst_sz, "%s", src);
}

static uint32_t get_timestamp_s(void)
{
    time_t now = time(NULL);

    if (now < 1704067200) {
        return 0;
    }

    return (uint32_t)now;
}

static void apply_startup_jitter(void)
{
    uint32_t span = STARTUP_JITTER_MAX_MS - STARTUP_JITTER_MIN_MS;
    uint32_t jitter = STARTUP_JITTER_MIN_MS;

    if (span > 0) {
        jitter += (esp_random() % (span + 1));
    }

    ESP_LOGI(TAG, "Aplicando startup jitter de %lu ms", (unsigned long)jitter);
    vTaskDelay(pdMS_TO_TICKS(jitter));
}

/* =====================================================
 * EVENTOS WIFI / IP / MESH
 * ===================================================== */
static void wifi_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != WIFI_EVENT) return;

    switch (id) {
    case WIFI_EVENT_STA_START:
        ESP_LOGI(TAG, "WIFI_EVENT_STA_START");
        break;

    case WIFI_EVENT_STA_CONNECTED:
        ESP_LOGI(TAG, "WIFI_EVENT_STA_CONNECTED");
        break;

    case WIFI_EVENT_STA_DISCONNECTED:
        ESP_LOGW(TAG, "WIFI desconectado (reconectando...)");
        esp_wifi_connect();
        break;

    default:
        break;
    }
}

static void ip_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != IP_EVENT || id != IP_EVENT_STA_GOT_IP) return;
    ip_event_got_ip_t *event = (ip_event_got_ip_t *)data;
    ESP_LOGI(TAG, "IP obtenida " IPSTR, IP2STR(&event->ip_info.ip));
}

static void mesh_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != MESH_EVENT) return;

    switch (id) {
    case MESH_EVENT_STARTED: {
        mesh_addr_t myid = {0};
        esp_mesh_get_id(&myid);
        ESP_LOGI(TAG, "Mesh iniciado ID=%02X:%02X:%02X:%02X:%02X:%02X",
                 myid.addr[0], myid.addr[1], myid.addr[2],
                 myid.addr[3], myid.addr[4], myid.addr[5]);
        break;
    }

    case MESH_EVENT_PARENT_CONNECTED:
        ESP_LOGI(TAG, "Parent conectado. layer=%d (root=%d)",
                 esp_mesh_get_layer(), esp_mesh_is_root());
        break;

    case MESH_EVENT_PARENT_DISCONNECTED:
        ESP_LOGW(TAG, "Parent desconectado. layer=%d", esp_mesh_get_layer());
        break;

    default:
        break;
    }
}

/* =====================================================
 * ENVÍO MESH V2
 * ===================================================== */
static esp_err_t mesh_send_msg(sensor_msg_v2_t *msg)
{
    mesh_data_t data = {
        .data  = (uint8_t *)msg,
        .size  = sizeof(sensor_msg_v2_t),
        .proto = MESH_PROTO_BIN,
        .tos   = MESH_TOS_P2P
    };

    return esp_mesh_send(NULL, &data, 0, NULL, 0);
}

static int mesh_send_metric_float(const char *metric_name, const char *unit, float value)
{
    mesh_addr_t parent;
    if (esp_mesh_get_parent_bssid(&parent) != ESP_OK) {
        ESP_LOGW(TAG, "No parent BSSID disponible para metric=%s", metric_name);
        return -1;
    }

    sensor_msg_v2_t msg;
    memset(&msg, 0, sizeof(msg));

    msg.sensor_type = SENSOR_DISTANCE;
    msg.value_type  = VALUE_FLOAT;
    msg.reserved    = 0;

    safe_copy_fixed(msg.node_name, sizeof(msg.node_name), NODE_NAME);
    safe_copy_fixed(msg.zone, sizeof(msg.zone), NODE_ZONE);
    safe_copy_fixed(msg.node_type, sizeof(msg.node_type), NODE_TYPE);
    safe_copy_fixed(msg.sensor_name, sizeof(msg.sensor_name), SENSOR_NAME);
    safe_copy_fixed(msg.metric_name, sizeof(msg.metric_name), metric_name);
    safe_copy_fixed(msg.unit, sizeof(msg.unit), unit);

    msg.v.f = value;
    msg.timestamp = get_timestamp_s();

    esp_err_t err = mesh_send_msg(&msg);
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "mesh_send FLOAT FAIL metric=%s err=%s",
                 metric_name, esp_err_to_name(err));
        return -1;
    }

    ESP_LOGI(TAG, "TX FLOAT metric=%s val=%.2f ts=%" PRIu32,
             metric_name, (double)value, msg.timestamp);
    return 0;
}

/* =====================================================
 * HC-SR04 GPIO INIT
 * ===================================================== */
static void hcsr04_gpio_init(void)
{
    gpio_config_t trig = {
        .pin_bit_mask = (1ULL << HCSR04_TRIG_GPIO),
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    ESP_ERROR_CHECK(gpio_config(&trig));
    gpio_set_level(HCSR04_TRIG_GPIO, 0);

    gpio_config_t echo = {
        .pin_bit_mask = (1ULL << HCSR04_ECHO_GPIO),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    ESP_ERROR_CHECK(gpio_config(&echo));

    ESP_LOGI(TAG, "HC-SR04 listo TRIG=GPIO%d ECHO=GPIO%d (ECHO debe ir a 3.3V)",
             HCSR04_TRIG_GPIO, HCSR04_ECHO_GPIO);
}

/* =====================================================
 * HC-SR04 MEDICIÓN
 * ===================================================== */
static bool wait_for_level(gpio_num_t pin, int level, int64_t timeout_us)
{
    int64_t start = esp_timer_get_time();
    while (gpio_get_level(pin) != level) {
        if ((esp_timer_get_time() - start) > timeout_us) {
            return false;
        }
        esp_rom_delay_us(5);
    }
    return true;
}

static bool hcsr04_read_distance_cm(float *out_cm)
{
    gpio_set_level(HCSR04_TRIG_GPIO, 0);
    esp_rom_delay_us(2);
    gpio_set_level(HCSR04_TRIG_GPIO, 1);
    esp_rom_delay_us(10);
    gpio_set_level(HCSR04_TRIG_GPIO, 0);

    if (!wait_for_level(HCSR04_ECHO_GPIO, 1, ECHO_WAIT_TIMEOUT_US)) {
        return false;
    }

    int64_t t_start = esp_timer_get_time();

    if (!wait_for_level(HCSR04_ECHO_GPIO, 0, ECHO_WAIT_TIMEOUT_US)) {
        return false;
    }

    int64_t t_end = esp_timer_get_time();
    int64_t pulse_us = t_end - t_start;

    float cm = (float)pulse_us * 0.0343f * 0.5f;

    if (cm < 0.0f || cm > MAX_DISTANCE_CM) {
        return false;
    }

    *out_cm = cm;
    return true;
}

/* =====================================================
 * TASK HC-SR04 -> MESH
 * ===================================================== */
static void hcsr04_task(void *arg)
{
    while (esp_mesh_get_layer() <= 0) {
        vTaskDelay(pdMS_TO_TICKS(200));
    }

    apply_startup_jitter();

    ESP_LOGI(TAG, "Nodo listo. layer=%d (root=%d). Enviando distancia cada %d ms...",
             esp_mesh_get_layer(), esp_mesh_is_root(), SEND_PERIOD_MS);

    while (1) {
        if (esp_mesh_get_layer() <= 0) {
            ESP_LOGW(TAG, "Sin layer (desconectado). Esperando reconexión...");
            while (esp_mesh_get_layer() <= 0) {
                vTaskDelay(pdMS_TO_TICKS(500));
            }
            ESP_LOGI(TAG, "Reconectado.");
        }

        float cm = 0.0f;
        bool ok = hcsr04_read_distance_cm(&cm);

        // Si falla, enviar distancia máxima 400cm
        float send_val = ok ? cm : MAX_DISTANCE_CM;

        mesh_send_metric_float("distance_cm", "cm", send_val);

        if (ok) {
            ESP_LOGI(TAG, "Distancia: %.2f cm (enviado)", (double)cm);
        } else {
            ESP_LOGW(TAG, "Lectura HC-SR04 falló (enviado distancia máxima: %.0f cm)", MAX_DISTANCE_CM);
        }

        vTaskDelay(pdMS_TO_TICKS(SEND_PERIOD_MS));
    }
}

/* =====================================================
 * APP MAIN
 * ===================================================== */
void app_main(void)
{
    ESP_ERROR_CHECK(nvs_flash_init());
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());

    ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(IP_EVENT, IP_EVENT_STA_GOT_IP, &ip_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(MESH_EVENT, ESP_EVENT_ANY_ID, &mesh_event_handler, NULL));

    esp_netif_t *netif_sta = NULL;
    esp_netif_t *netif_ap = NULL;
    ESP_ERROR_CHECK(esp_netif_create_default_wifi_mesh_netifs(&netif_sta, &netif_ap));

    wifi_init_config_t wifi_cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&wifi_cfg));
    ESP_ERROR_CHECK(esp_wifi_set_storage(WIFI_STORAGE_FLASH));
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_APSTA));
    ESP_ERROR_CHECK(esp_wifi_set_ps(WIFI_PS_NONE));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_ERROR_CHECK(esp_mesh_init());

    mesh_cfg_t mc = MESH_INIT_CONFIG_DEFAULT();
    uint8_t mid[6] = MESH_ID;
    memcpy(mc.mesh_id.addr, mid, 6);

    mc.channel = ROUTER_CH;

    mc.router.ssid_len = strlen(ROUTER_SSID);
    memcpy(mc.router.ssid, ROUTER_SSID, mc.router.ssid_len);
    memcpy(mc.router.password, ROUTER_PASS, strlen(ROUTER_PASS));

    mc.mesh_ap.max_connection = MESH_MAX_CONN;
    mc.mesh_ap.nonmesh_max_connection = 0;
    memcpy(mc.mesh_ap.password, MESH_AP_PASS, strlen(MESH_AP_PASS));

    ESP_ERROR_CHECK(esp_mesh_set_config(&mc));
    ESP_ERROR_CHECK(esp_mesh_set_self_organized(true, false));
    ESP_ERROR_CHECK(esp_mesh_start());

    ESP_LOGI(TAG, "Nodo HC-SR04 iniciado. NO ROOT");
    ESP_LOGI(TAG, "SEND_PERIOD_MS=%d", SEND_PERIOD_MS);

    hcsr04_gpio_init();

    xTaskCreate(hcsr04_task, "hcsr04_task", 4096, NULL, 5, NULL);
}
