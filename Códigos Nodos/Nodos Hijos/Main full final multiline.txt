/* ================= STRUCT ================= */
/*
 * IMPORTANTE:
 * Los nodos hijos deben usar EXACTAMENTE este mismo struct.
 *
 * value_type:
 *   0 = int32
 *   1 = float
 */
typedef struct __attribute__((packed)) {
    uint8_t value_type;
    char node_name[24];
    char zone[24];
    char sensor[24];
    char metric[16];
    char unit[16];
    union {
        int32_t i;
        float f;
    } v;
    uint32_t timestamp;
} msg_t;

typedef struct {
    uint8_t mac[6];
    msg_t msg;
} item_t;

/* ================= HELPERS ================= */

static void sanitize(char *dst, const char *src, size_t size)
{
    if (!dst || size == 0) return;

    dst[0] = '\0';
    if (!src) return;

    size_t j = 0;
    for (size_t i = 0; src[i] != '\0' && j < size - 1; i++) {
        char c = src[i];
        if (c == ' ' || c == ',' || c == '=' || c == '\n' || c == '\r' || c == '\t') {
            dst[j++] = '_';
        } else if ((unsigned char)c < 32 || (unsigned char)c > 126) {
            dst[j++] = '_';
        } else {
            dst[j++] = c;
        }
    }
    dst[j] = '\0';
}

static bool is_valid_unix_ts(uint32_t ts)
{
    return (ts >= 1700000000UL);
}

static void update_queue_peak(void)
{
    if (!queue) return;

    UBaseType_t used = uxQueueMessagesWaiting(queue);
    if ((uint32_t)used > g_queue_peak) {
        g_queue_peak = (uint32_t)used;
    }
}

static void requeue_batch(item_t *batch, int count)
{
    if (!batch || count <= 0) return;

    for (int i = count - 1; i >= 0; i--) {
        if (xQueueSendToFront(queue, &batch[i], pdMS_TO_TICKS(10)) == pdTRUE) {
            g_requeued++;
        } else {
            g_queue_drops++;
            ESP_LOGE(TAG, "No se pudo re-encolar item %d", i);
        }
    }

    update_queue_peak();
}

static bool build_line_protocol(const item_t *item, char *line, size_t line_size)
{
    if (!item || !line || line_size == 0) return false;

    char measurement[48];
    char node[32], zone[32], sensor[32], metric[24], unit[24];

    sanitize(measurement, INFLUX_MEASUREMENT, sizeof(measurement));
    sanitize(node, item->msg.node_name, sizeof(node));
    sanitize(zone, item->msg.zone, sizeof(zone));
    sanitize(sensor, item->msg.sensor, sizeof(sensor));
    sanitize(metric, item->msg.metric, sizeof(metric));
    sanitize(unit, item->msg.unit, sizeof(unit));

    int written = 0;

    if (item->msg.value_type == 0) {
        if (is_valid_unix_ts(item->msg.timestamp)) {
            written = snprintf(
                line, line_size,
                "%s,node=%s,zone=%s,sensor=%s,metric=%s,unit=%s value_i=%ldi %lu",
                measurement, node, zone, sensor, metric, unit,
                (long)item->msg.v.i,
                (unsigned long)item->msg.timestamp
            );
        } else {
            written = snprintf(
                line, line_size,
                "%s,node=%s,zone=%s,sensor=%s,metric=%s,unit=%s value_i=%ldi",
                measurement, node, zone, sensor, metric, unit,
                (long)item->msg.v.i
            );
        }
    } else if (item->msg.value_type == 1) {
        if (is_valid_unix_ts(item->msg.timestamp)) {
            written = snprintf(
                line, line_size,
                "%s,node=%s,zone=%s,sensor=%s,metric=%s,unit=%s value_f=%.3f %lu",
                measurement, node, zone, sensor, metric, unit,
                item->msg.v.f,
                (unsigned long)item->msg.timestamp
            );
        } else {
            written = snprintf(
                line, line_size,
                "%s,node=%s,zone=%s,sensor=%s,metric=%s,unit=%s value_f=%.3f",
                measurement, node, zone, sensor, metric, unit,
                item->msg.v.f
            );
        }
    } else {
        ESP_LOGW(TAG, "value_type desconocido: %u", item->msg.value_type);
        return false;
    }

    return (written > 0 && written < (int)line_size);
}

/* ================= EVENTOS ================= */

static void wifi_event_handler(void *arg,
                               esp_event_base_t base,
                               int32_t id,
                               void *data)
{
    if (base == WIFI_EVENT && id == WIFI_EVENT_STA_DISCONNECTED) {
        has_ip = false;
        ESP_LOGW(TAG, "WiFi STA desconectado");
    }
}

static void mesh_event_handler(void *arg,
                               esp_event_base_t base,
                               int32_t id,
                               void *data)
{
    switch (id) {
        case MESH_EVENT_STARTED:
            ESP_LOGI(TAG, "MESH STARTED");
            break;

        case MESH_EVENT_STOPPED:
            ESP_LOGW(TAG, "MESH STOPPED");
            break;

        case MESH_EVENT_CHILD_CONNECTED: {
            mesh_event_child_connected_t *e = (mesh_event_child_connected_t *)data;
            ESP_LOGI(TAG, "CHILD CONNECTED: %02X:%02X:%02X:%02X:%02X:%02X aid=%d",
                     e->mac[0], e->mac[1], e->mac[2],
                     e->mac[3], e->mac[4], e->mac[5], e->aid);
            break;
        }

        case MESH_EVENT_CHILD_DISCONNECTED: {
            mesh_event_child_disconnected_t *e = (mesh_event_child_disconnected_t *)data;
            ESP_LOGW(TAG, "CHILD DISCONNECTED: %02X:%02X:%02X:%02X:%02X:%02X aid=%d",
                     e->mac[0], e->mac[1], e->mac[2],
                     e->mac[3], e->mac[4], e->mac[5], e->aid);
            break;
        }

        case MESH_EVENT_PARENT_CONNECTED: {
            mesh_event_connected_t *e = (mesh_event_connected_t *)data;
            ESP_LOGI(TAG, "PARENT CONNECTED layer=%d parent=%02X:%02X:%02X:%02X:%02X:%02X",
                     e->self_layer,
                     e->connected.bssid[0], e->connected.bssid[1], e->connected.bssid[2],
                     e->connected.bssid[3], e->connected.bssid[4], e->connected.bssid[5]);
            break;
        }

        case MESH_EVENT_PARENT_DISCONNECTED: {
            mesh_event_disconnected_t *e = (mesh_event_disconnected_t *)data;
            ESP_LOGW(TAG, "PARENT DISCONNECTED reason=%d", e->reason);
            has_ip = false;
            break;
        }

        default:
            break;
    }
}

static void ip_event_handler(void *arg,
                             esp_event_base_t base,
                             int32_t id,
                             void *data)
{
    if (base == IP_EVENT && id == IP_EVENT_STA_GOT_IP) {
        ip_event_got_ip_t *event = (ip_event_got_ip_t *)data;
        has_ip = true;
        ESP_LOGI(TAG, "IP OK: " IPSTR, IP2STR(&event->ip_info.ip));
    }
}

/* ================= RX ================= */

static void rx_task(void *arg)
{
    mesh_addr_t from;
    mesh_data_t data;
    uint8_t buf[RX_BUF_SIZE];
    int flag = 0;

    while (1) {
        memset(&from, 0, sizeof(from));
        memset(&data, 0, sizeof(data));
        memset(buf, 0, sizeof(buf));

        data.data = buf;
        data.size = sizeof(buf);

        esp_err_t err = esp_mesh_recv(&from, &data, portMAX_DELAY, &flag, NULL, 0);
        if (err != ESP_OK) {
            ESP_LOGW(TAG, "esp_mesh_recv error: %s", esp_err_to_name(err));
            continue;
        }

        if (data.size != sizeof(msg_t)) {
            g_invalid_size++;
            ESP_LOGW(TAG, "Tamaño inválido recibido: %d esperado: %u",
                     data.size, (unsigned)sizeof(msg_t));
            continue;
        }

        item_t item;
        memset(&item, 0, sizeof(item));
        memcpy(item.mac, from.addr, 6);
        memcpy(&item.msg, buf, sizeof(msg_t));

        g_rx_msgs++;
        if (item.msg.value_type == 0) g_rx_int++;
        else if (item.msg.value_type == 1) g_rx_float++;

        if (item.msg.value_type == 0) {
            ESP_LOGI(TAG, "RX node=%s zone=%s sensor=%s metric=%s unit=%s int=%ld ts=%lu",
                     item.msg.node_name, item.msg.zone, item.msg.sensor,
                     item.msg.metric, item.msg.unit,
                     (long)item.msg.v.i,
                     (unsigned long)item.msg.timestamp);
        } else {
            ESP_LOGI(TAG, "RX node=%s zone=%s sensor=%s metric=%s unit=%s float=%.3f ts=%lu",
                     item.msg.node_name, item.msg.zone, item.msg.sensor,
                     item.msg.metric, item.msg.unit,
                     item.msg.v.f,
                     (unsigned long)item.msg.timestamp);
        }

        if (xQueueSend(queue, &item, pdMS_TO_TICKS(10)) != pdTRUE) {
            g_queue_drops++;
            ESP_LOGW(TAG, "Cola llena");
        }

        update_queue_peak();
    }
}

/* ================= HTTP/INFLUX ================= */

static bool send_payload_to_influx(const char *url, const char *payload, int len)
{
    if (!url || !payload || len <= 0) return false;

    for (int r = 0; r < RETRY_COUNT; r++) {
        esp_http_client_config_t cfg = {
            .url = url,
            .method = HTTP_METHOD_POST,
            .timeout_ms = HTTP_TIMEOUT_MS,
            .keep_alive_enable = false,
        };

        esp_http_client_handle_t client = esp_http_client_init(&cfg);
        if (!client) {
            ESP_LOGE(TAG, "No se pudo crear cliente HTTP");
            return false;
        }

        char auth[320];
        snprintf(auth, sizeof(auth), "Token %s", INFLUX_TOKEN);

        ESP_LOGD(TAG, "Auth header: %s", auth);
        
        esp_http_client_set_header(client, "Authorization", auth);
        esp_http_client_set_header(client, "Content-Type", "text/plain; charset=utf-8");
        esp_http_client_set_post_field(client, payload, len);

        esp_err_t err = esp_http_client_perform(client);
        int status = esp_http_client_get_status_code(client);

        ESP_LOGI(TAG, "POST intento %d err=%s status=%d",
                 r + 1, esp_err_to_name(err), status);

        // Solo intentar leer respuesta si no hay error en la solicitud
        if (err == ESP_OK) {
            char resp[512] = {0};
            int read_len = 0;
            
            // Intentar leer respuesta con manejo seguro
            if (status != 204) {  // 204 No Content no tiene body
                read_len = esp_http_client_read_response(client, resp, sizeof(resp) - 1);
                if (read_len > 0 && read_len < (int)sizeof(resp)) {
                    resp[read_len] = '\0';
                    ESP_LOGW(TAG, "Response (%d bytes): %s", read_len, resp);
                }
            }
        } else {
            ESP_LOGE(TAG, "HTTP request failed: %s", esp_err_to_name(err));
        }

        esp_http_client_cleanup(client);

        if (err == ESP_OK && (status == 200 || status == 204)) {
            return true;
        }
    }

    return false;
}

static void influx_task(void *arg)
{
    while (!has_ip) {
        vTaskDelay(pdMS_TO_TICKS(100));
    }

    while (1) {
        memset(g_batch, 0, sizeof(g_batch));
        memset(g_payload, 0, sizeof(g_payload));

        int count = 0;

        if (xQueueReceive(queue, &g_batch[count], portMAX_DELAY) != pdTRUE) {
            continue;
        }
        count++;

        TickType_t start = xTaskGetTickCount();
        while (count < BATCH_MAX &&
               (xTaskGetTickCount() - start) < pdMS_TO_TICKS(BATCH_WAIT_MS)) {
            if (xQueueReceive(queue, &g_batch[count], pdMS_TO_TICKS(5)) == pdTRUE) {
                count++;
            }
        }

        if (!has_ip) {
            ESP_LOGW(TAG, "Sin IP, re-encolando");
            requeue_batch(g_batch, count);
            vTaskDelay(pdMS_TO_TICKS(500));
            continue;
        }

        int len = 0;
        int valid = 0;

        for (int i = 0; i < count; i++) {
            char line[256];

            if (!build_line_protocol(&g_batch[i], line, sizeof(line))) {
                ESP_LOGW(TAG, "No se pudo construir línea LP para item %d", i);
                continue;
            }

            int l = strlen(line);
            int extra = (valid > 0) ? 1 : 0;

            if ((len + extra + l) >= PAYLOAD_SIZE) {
                ESP_LOGW(TAG, "Payload lleno, batch truncado");
                break;
            }

            if (valid > 0) {
                g_payload[len++] = '\n';
            }

            memcpy(g_payload + len, line, l);
            len += l;
            valid++;
        }

        if (valid == 0) {
            ESP_LOGW(TAG, "Batch vacío");
            continue;
        }

        g_payload[len] = '\0';

        char url[256];
        snprintf(url, sizeof(url),
                 "%s/api/v2/write?org=%s&bucket=%s&precision=s",
                 INFLUX_URL, INFLUX_ORG, INFLUX_BUCKET);

        ESP_LOGI(TAG, "=========== BATCH A INFLUX ===========");
        ESP_LOGI(TAG, "Write URL: %s", url);
        ESP_LOGI(TAG, "Measurement: %s", INFLUX_MEASUREMENT);
        ESP_LOGI(TAG, "Items batch: %d | lineas validas: %d | bytes: %d", count, valid, len);
        ESP_LOGI(TAG, "Payload:\n%s", g_payload);
        ESP_LOGI(TAG, "======================================");

        bool ok = send_payload_to_influx(url, g_payload, len);

        if (ok) {
            g_sent_batches++;
            g_sent_points += valid;
            ESP_LOGI(TAG, "Batch enviado OK (%d)", valid);
        } else {
            g_http_fail++;
            ESP_LOGE(TAG, "Error HTTP, re-encolando batch");
            requeue_batch(g_batch, count);
        }

        update_queue_peak();
    }
}

/* ================= STATS ================= */

static void stats_task(void *arg)
{
    uint32_t last_rx = 0;
    uint32_t last_sent = 0;

    while (1) {
        uint32_t rx_now = g_rx_msgs;
        uint32_t sent_now = g_sent_points;

        uint32_t rx_rate = rx_now - last_rx;
        uint32_t sent_rate = sent_now - last_sent;

        last_rx = rx_now;
        last_sent = sent_now;

        UBaseType_t q_now = queue ? uxQueueMessagesWaiting(queue) : 0;

        ESP_LOGI(TAG,
                 "CAPACITY | rx_total=%lu rx_rate=%lu/s sent_total=%lu sent_rate=%lu/s "
                 "batches=%lu requeued=%lu http_fail=%lu q_now=%lu q_peak=%lu q_drop=%lu invalid=%lu int=%lu float=%lu",
                 (unsigned long)g_rx_msgs,
                 (unsigned long)rx_rate,
                 (unsigned long)g_sent_points,
                 (unsigned long)sent_rate,
                 (unsigned long)g_sent_batches,
                 (unsigned long)g_requeued,
                 (unsigned long)g_http_fail,
                 (unsigned long)q_now,
                 (unsigned long)g_queue_peak,
                 (unsigned long)g_queue_drops,
                 (unsigned long)g_invalid_size,
                 (unsigned long)g_rx_int,
                 (unsigned long)g_rx_float);

        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

/* ================= MAIN ================= */

void app_main(void)
{
    esp_err_t err = nvs_flash_init();
    if (err == ESP_ERR_NVS_NO_FREE_PAGES || err == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        err = nvs_flash_init();
    }
    ESP_ERROR_CHECK(err);

    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());

    queue = xQueueCreate(RX_QUEUE_LEN, sizeof(item_t));
    if (!queue) {
        ESP_LOGE(TAG, "No se pudo crear la cola");
        return;
    }

    ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID, wifi_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(MESH_EVENT, ESP_EVENT_ANY_ID, mesh_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(IP_EVENT, IP_EVENT_STA_GOT_IP, ip_event_handler, NULL));

    esp_netif_create_default_wifi_sta();
    esp_netif_create_default_wifi_ap();

    wifi_init_config_t wifi_cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&wifi_cfg));
    ESP_ERROR_CHECK(esp_wifi_set_storage(WIFI_STORAGE_FLASH));
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_APSTA));
    ESP_ERROR_CHECK(esp_wifi_set_ps(WIFI_PS_NONE));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_ERROR_CHECK(esp_mesh_init());

    mesh_cfg_t mc = MESH_INIT_CONFIG_DEFAULT();
    uint8_t id[6] = MESH_ID;
    memcpy(mc.mesh_id.addr, id, 6);

    mc.channel = 0;

    mc.router.ssid_len = strlen(ROUTER_SSID);
    memcpy(mc.router.ssid, ROUTER_SSID, mc.router.ssid_len);
    memcpy(mc.router.password, ROUTER_PASS, strlen(ROUTER_PASS));

    mc.mesh_ap.max_connection = MESH_MAX_CONN;
    mc.mesh_ap.nonmesh_max_connection = 0;
    memcpy(mc.mesh_ap.password, MESH_AP_PASS, strlen(MESH_AP_PASS));

    ESP_ERROR_CHECK(esp_mesh_set_config(&mc));
    ESP_ERROR_CHECK(esp_mesh_set_type(MESH_ROOT));
    ESP_ERROR_CHECK(esp_mesh_start());

    xTaskCreate(rx_task, "rx_task", 4096, NULL, 6, NULL);
    xTaskCreate(influx_task, "influx_task", 12288, NULL, 5, NULL);
    xTaskCreate(stats_task, "stats_task", 4096, NULL, 4, NULL);

    ESP_LOGI(TAG, "ROOT OK");
}
