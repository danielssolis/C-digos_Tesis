#define SENSOR_NAME_DHT11  "dht11_Ducha"
#define SENSOR_NAME_YF201  "yf_s201_Ducha"

/* =========================
 * PROTOCOLO ROOT ACTUAL
 * ========================= */
typedef enum {
    VALUE_TYPE_INT   = 0,
    VALUE_TYPE_FLOAT = 1
} value_type_t;

typedef struct __attribute__((packed)) {
    uint8_t value_type;
    char node_name[24];
    char zone[24];
    char sensor[24];
    char metric[16];
    char unit[16];
    union {
        int32_t i;
        float   f;
    } v;
    uint32_t timestamp;
} msg_t;

/* =========================
 * PINES
 * ========================= */
#define HCSR04_TRIG_GPIO       19
#define HCSR04_ECHO_GPIO       18
#define DHT11_GPIO             17
#define YF_S201_GPIO           16

/* =========================
 * TEMPORIZACIÓN
 * ========================= */
#define SEND_PERIOD_MS           5000
#define STARTUP_JITTER_MIN_MS    1000
#define STARTUP_JITTER_MAX_MS    5000
#define HCSR04_TIMEOUT_US        20100
#define MIN_PULSE_INTERVAL_US    20000
#define HCSR04_CHANGE_THRESHOLD  60.0f
#define HCSR04_MAX_DISTANCE_CM   400.0f

/* =========================
 * GLOBALES YF-S201
 * ========================= */
static volatile uint32_t flow_pulses_s201 = 0;
static volatile int64_t  last_pulse_s201_us = 0;

/* =====================================================
 * HELPERS
 * ===================================================== */
static void safe_copy_fixed(char *dst, size_t dst_sz, const char *src)
{
    if (!dst || dst_sz == 0) return;
    memset(dst, 0, dst_sz);
    if (!src) return;
    snprintf(dst, dst_sz, "%s", src);
}

static uint32_t get_timestamp_s(void)
{
    time_t now = time(NULL);

    if ((uint32_t)now >= 1700000000UL) {
        return (uint32_t)now;
    }

    return (uint32_t)(esp_timer_get_time() / 1000000ULL);
}

static void apply_startup_jitter(void)
{
    uint32_t span = STARTUP_JITTER_MAX_MS - STARTUP_JITTER_MIN_MS;
    uint32_t jitter = STARTUP_JITTER_MIN_MS;

    if (span > 0) {
        jitter += (esp_random() % (span + 1));
    }

    ESP_LOGI(TAG, "Aplicando startup jitter de %lu ms", (unsigned long)jitter);
    vTaskDelay(pdMS_TO_TICKS(jitter));
}

/* =====================================================
 * EVENTOS WIFI / IP / MESH
 * ===================================================== */
static void wifi_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != WIFI_EVENT) return;

    switch (id) {
        case WIFI_EVENT_STA_START:
            ESP_LOGI(TAG, "WIFI_EVENT_STA_START");
            break;
        case WIFI_EVENT_STA_CONNECTED:
            ESP_LOGI(TAG, "WIFI_EVENT_STA_CONNECTED");
            break;
        case WIFI_EVENT_STA_DISCONNECTED:
            ESP_LOGW(TAG, "WIFI desconectado");
            break;
        default:
            break;
    }
}

static void ip_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != IP_EVENT || id != IP_EVENT_STA_GOT_IP) return;

    ip_event_got_ip_t *event = (ip_event_got_ip_t *)data;
    ESP_LOGI(TAG, "IP obtenida " IPSTR, IP2STR(&event->ip_info.ip));
}

static void mesh_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != MESH_EVENT) return;

    switch (id) {
        case MESH_EVENT_STARTED: {
            mesh_addr_t myid = {0};
            esp_mesh_get_id(&myid);
            ESP_LOGI(TAG, "Mesh iniciado ID=%02X:%02X:%02X:%02X:%02X:%02X",
                     myid.addr[0], myid.addr[1], myid.addr[2],
                     myid.addr[3], myid.addr[4], myid.addr[5]);
            break;
        }

        case MESH_EVENT_PARENT_CONNECTED:
            ESP_LOGI(TAG, "Parent conectado. layer=%d root=%d",
                     esp_mesh_get_layer(), esp_mesh_is_root());
            break;

        case MESH_EVENT_PARENT_DISCONNECTED:
            ESP_LOGW(TAG, "Parent desconectado. layer=%d",
                     esp_mesh_get_layer());
            break;

        default:
            break;
    }
}

/* =====================================================
 * ENVÍO MESH
 * ===================================================== */
static esp_err_t mesh_send_msg(msg_t *msg)
{
    mesh_data_t data = {
        .data  = (uint8_t *)msg,
        .size  = sizeof(msg_t),
        .proto = MESH_PROTO_BIN,
        .tos   = MESH_TOS_P2P
    };

    return esp_mesh_send(NULL, &data, 0, NULL, 0);
}

static int mesh_send_metric_int(const char *sensor_name, const char *metric_name, const char *unit_name, int32_t value)
{
    if (esp_mesh_get_layer() <= 0) return -1;

    msg_t msg;
    memset(&msg, 0, sizeof(msg));

    msg.value_type = VALUE_TYPE_INT;

    safe_copy_fixed(msg.node_name, sizeof(msg.node_name), NODE_NAME);
    safe_copy_fixed(msg.zone, sizeof(msg.zone), NODE_ZONE);
    safe_copy_fixed(msg.sensor, sizeof(msg.sensor), sensor_name);
    safe_copy_fixed(msg.metric, sizeof(msg.metric), metric_name);
    safe_copy_fixed(msg.unit, sizeof(msg.unit), unit_name);

    msg.v.i = value;
    msg.timestamp = get_timestamp_s();

    esp_err_t err = mesh_send_msg(&msg);
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "mesh_send INT FAIL sensor=%s metric=%s err=%s",
                 sensor_name, metric_name, esp_err_to_name(err));
        return -1;
    }

    return 0;
}

static int mesh_send_metric_float(const char *sensor_name, const char *metric_name, const char *unit_name, float value)
{
    if (esp_mesh_get_layer() <= 0) return -1;

    msg_t msg;
    memset(&msg, 0, sizeof(msg));

    msg.value_type = VALUE_TYPE_FLOAT;

    safe_copy_fixed(msg.node_name, sizeof(msg.node_name), NODE_NAME);
    safe_copy_fixed(msg.zone, sizeof(msg.zone), NODE_ZONE);
    safe_copy_fixed(msg.sensor, sizeof(msg.sensor), sensor_name);
    safe_copy_fixed(msg.metric, sizeof(msg.metric), metric_name);
    safe_copy_fixed(msg.unit, sizeof(msg.unit), unit_name);

    msg.v.f = value;
    msg.timestamp = get_timestamp_s();

    esp_err_t err = mesh_send_msg(&msg);
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "mesh_send FLOAT FAIL sensor=%s metric=%s err=%s",
                 sensor_name, metric_name, esp_err_to_name(err));
        return -1;
    }

    return 0;
}

/* =====================================================
 * 1) HC-SR04
 * ===================================================== */
static void hcsr04_init(void)
{
    gpio_config_t trig_conf = {
        .pin_bit_mask = 1ULL << HCSR04_TRIG_GPIO,
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    ESP_ERROR_CHECK(gpio_config(&trig_conf));
    gpio_set_level(HCSR04_TRIG_GPIO, 0);

    gpio_config_t echo_conf = {
        .pin_bit_mask = 1ULL << HCSR04_ECHO_GPIO,
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    ESP_ERROR_CHECK(gpio_config(&echo_conf));

    ESP_LOGI(TAG, "HC-SR04 TRIG=%d ECHO=%d listo", HCSR04_TRIG_GPIO, HCSR04_ECHO_GPIO);
}

static bool wait_for_level_hcsr04(gpio_num_t pin, int level, int64_t timeout_us)
{
    int64_t start = esp_timer_get_time();
    while (gpio_get_level(pin) != level) {
        if ((esp_timer_get_time() - start) > timeout_us) {
            return false;
        }
        esp_rom_delay_us(5);
    }
    return true;
}

static bool hcsr04_read_distance_cm(float *out_cm)
{
    // Pulso TRIG 10us
    gpio_set_level(HCSR04_TRIG_GPIO, 0);
    esp_rom_delay_us(2);
    gpio_set_level(HCSR04_TRIG_GPIO, 1);
    esp_rom_delay_us(10);
    gpio_set_level(HCSR04_TRIG_GPIO, 0);

    // Esperar ECHO HIGH
    if (!wait_for_level_hcsr04(HCSR04_ECHO_GPIO, 1, HCSR04_TIMEOUT_US)) {
        return false;
    }

    int64_t t_start = esp_timer_get_time();

    // Esperar ECHO LOW
    if (!wait_for_level_hcsr04(HCSR04_ECHO_GPIO, 0, HCSR04_TIMEOUT_US)) {
        return false;
    }

    int64_t t_end = esp_timer_get_time();
    int64_t pulse_us = t_end - t_start;

    float cm = (float)pulse_us * 0.0343f * 0.5f;

    if (cm < 0.0f || cm > HCSR04_MAX_DISTANCE_CM) {
        return false;
    }

    *out_cm = cm;
    return true;
}

static void hcsr04_task(void *arg)
{
    while (esp_mesh_get_layer() <= 0) {
        vTaskDelay(pdMS_TO_TICKS(200));
    }

    apply_startup_jitter();
    ESP_LOGI(TAG, "HC-SR04 activo con detección de cambios (período %d ms)", SEND_PERIOD_MS);

    float last_send_value = -1.0f;
    int64_t last_send_time = 0;

    while (1) {
        if (esp_mesh_get_layer() <= 0) {
            ESP_LOGW(TAG, "HC-SR04: Sin mesh. Esperando reconexión...");
            while (esp_mesh_get_layer() <= 0) {
                vTaskDelay(pdMS_TO_TICKS(500));
            }
            ESP_LOGI(TAG, "HC-SR04: Reconectado.");
        }

        float cm = 0.0f;
        bool ok = hcsr04_read_distance_cm(&cm);

        // Si falla lectura, envía 0 (sin detección)
        float send_val = ok ? cm : 0.0f;

        int64_t now = esp_timer_get_time() / 1000LL;  // Tiempo en ms
        bool should_send = false;
        const char *send_reason = "periódico";

        // Condición 1: Cambio vuelto a cero (de lectura válida a error)
        if (last_send_value > 0.0f && send_val == 0.0f) {
            should_send = true;
            send_reason = "transición a 0";
            ESP_LOGW(TAG, "HC-SR04: Transición a 0 detectada: %.2f -> 0 cm", last_send_value);
        }

        // Condición 2: Cambio superior a umbral
        if (last_send_value >= 0.0f && fabs(send_val - last_send_value) >= HCSR04_CHANGE_THRESHOLD) {
            should_send = true;
            send_reason = "cambio >= 60cm";
            ESP_LOGW(TAG, "HC-SR04: Cambio grande detectado: %.2f -> %.2f cm (diff: %.2f)",
                     last_send_value, send_val, fabs(send_val - last_send_value));
        }

        // Condición 3: Período de envío cumplido
        if ((now - last_send_time) >= SEND_PERIOD_MS) {
            should_send = true;
            send_reason = "periódico";
        }

        if (should_send) {
            mesh_send_metric_float(SENSOR_NAME_HCSR04, "distance", "cm", send_val);
            last_send_value = send_val;
            last_send_time = now;

            if (ok) {
                ESP_LOGI(TAG, "HC-SR04: Distancia %.2f cm [%s]", (double)cm, send_reason);
            } else {
                ESP_LOGW(TAG, "HC-SR04: Lectura falló, enviado 0 [%s]", send_reason);
            }
        }

        vTaskDelay(pdMS_TO_TICKS(100));  // Poll más frecuente para detectar cambios rápidamente
    }
}

/* =====================================================
 * 2) DHT11
 * ===================================================== */
static void dht11_init(void)
{
    gpio_config_t dht_conf = {
        .pin_bit_mask = 1ULL << DHT11_GPIO,
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    ESP_ERROR_CHECK(gpio_config(&dht_conf));
    gpio_set_level(DHT11_GPIO, 1);

    ESP_LOGI(TAG, "DHT11 listo GPIO%d", DHT11_GPIO);
}

static bool dht11_read(float *temp, float *hum)
{
    uint8_t data[5] = {0};
    int64_t start_time;

    gpio_set_direction(DHT11_GPIO, GPIO_MODE_OUTPUT);
    gpio_set_level(DHT11_GPIO, 0);
    vTaskDelay(pdMS_TO_TICKS(20));
    gpio_set_level(DHT11_GPIO, 1);
    esp_rom_delay_us(30);
    gpio_set_direction(DHT11_GPIO, GPIO_MODE_INPUT);

    start_time = esp_timer_get_time();
    while (gpio_get_level(DHT11_GPIO) == 1) {
        if ((esp_timer_get_time() - start_time) > 100) return false;
    }

    start_time = esp_timer_get_time();
    while (gpio_get_level(DHT11_GPIO) == 0) {
        if ((esp_timer_get_time() - start_time) > 100) return false;
    }

    start_time = esp_timer_get_time();
    while (gpio_get_level(DHT11_GPIO) == 1) {
        if ((esp_timer_get_time() - start_time) > 100) return false;
    }

    for (int i = 0; i < 40; i++) {
        start_time = esp_timer_get_time();
        while (gpio_get_level(DHT11_GPIO) == 0) {
            if ((esp_timer_get_time() - start_time) > 70) return false;
        }

        start_time = esp_timer_get_time();
        while (gpio_get_level(DHT11_GPIO) == 1) {
            if ((esp_timer_get_time() - start_time) > 120) return false;
        }

        int duration = (int)(esp_timer_get_time() - start_time);
        data[i / 8] <<= 1;
        if (duration > 40) data[i / 8] |= 1;
    }

    gpio_set_direction(DHT11_GPIO, GPIO_MODE_OUTPUT);
    gpio_set_level(DHT11_GPIO, 1);

    uint8_t checksum = data[0] + data[1] + data[2] + data[3];
    if (checksum != data[4]) {
        ESP_LOGW(TAG, "DHT11 checksum inválido (%02X vs %02X)", checksum, data[4]);
        return false;
    }

    *hum = data[0];
    *temp = data[2];
    return true;
}

static void dht11_task(void *arg)
{
    ESP_LOGI(TAG, "DHT11 activo (cada %d ms)", SEND_PERIOD_MS);
    vTaskDelay(pdMS_TO_TICKS(2000));

    while (1) {
        if (esp_mesh_get_layer() <= 0) {
            vTaskDelay(pdMS_TO_TICKS(1000));
            continue;
        }

        float temp = 0.0f, hum = 0.0f;
        if (dht11_read(&temp, &hum)) {
            mesh_send_metric_float(SENSOR_NAME_DHT11, "temperature", "celsius", temp);
            vTaskDelay(pdMS_TO_TICKS(100));
            mesh_send_metric_float(SENSOR_NAME_DHT11, "humidity", "percent", hum);

            ESP_LOGI(TAG, "DHT11 TEMP=%.1f C | HUM=%.1f %%", (double)temp, (double)hum);
        } else {
            ESP_LOGW(TAG, "DHT11 lectura fallida");
        }

        vTaskDelay(pdMS_TO_TICKS(SEND_PERIOD_MS));
    }
}

/* =====================================================
 * 3) YF-S201
 * Conserva lógica ON/OFF de la v1
 * y agrega:
 * - envío inmediato al cambio
 * - envío periódico cada 5 s
 * ===================================================== */
static void IRAM_ATTR flow_isr_yf201_handler(void *arg)
{
    int64_t now = esp_timer_get_time();
    if (last_pulse_s201_us != 0 && (now - last_pulse_s201_us) < MIN_PULSE_INTERVAL_US) return;
    last_pulse_s201_us = now;
    flow_pulses_s201++;
}

static void yf_s201_init(void)
{
    gpio_config_t io_conf = {
        .pin_bit_mask = 1ULL << YF_S201_GPIO,
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_POSEDGE
    };
    ESP_ERROR_CHECK(gpio_config(&io_conf));

    ESP_ERROR_CHECK(gpio_install_isr_service(0));
    ESP_ERROR_CHECK(gpio_isr_handler_add(YF_S201_GPIO, flow_isr_yf201_handler, NULL));

    ESP_LOGI(TAG, "YF-S201 listo GPIO%d", YF_S201_GPIO);
}

static void yf_s201_task(void *arg)
{
    int last_flow_state = -1;
    int current_flow_state = 0;
    uint32_t elapsed_ms = 0;
    bool first_send = true;

    ESP_LOGI(TAG, "YF-S201 ON/OFF activo");

    while (1) {
        if (esp_mesh_get_layer() <= 0) {
            vTaskDelay(pdMS_TO_TICKS(1000));
            continue;
        }

        uint32_t pulses = flow_pulses_s201;
        flow_pulses_s201 = 0;

        int64_t now_us = esp_timer_get_time();
        int64_t last_us = last_pulse_s201_us;
        bool flow_on = false;

        if (pulses >= 5) {
            flow_on = true;
        } else if (last_us != 0) {
            int64_t dt_ms = (now_us - last_us) / 1000;
            flow_on = (dt_ms <= 2000);
        }

        current_flow_state = flow_on ? 1 : 0;

        if (first_send || current_flow_state != last_flow_state) {
            mesh_send_metric_int(SENSOR_NAME_YF201, "flow_state", "bool", current_flow_state);
            ESP_LOGI(TAG, "YF-S201 CAMBIO -> Pulsos=%" PRIu32 " FLUJO=%s",
                     pulses, flow_on ? "ON" : "OFF");
            last_flow_state = current_flow_state;
            first_send = false;
            elapsed_ms = 0;
        } else {
            elapsed_ms += 1000;
        }

        if (elapsed_ms >= SEND_PERIOD_MS) {
            mesh_send_metric_int(SENSOR_NAME_YF201, "flow_state", "bool", current_flow_state);
            ESP_LOGI(TAG, "YF-S201 PERIODICO -> Pulsos=%" PRIu32 " FLUJO=%s",
                     pulses, flow_on ? "ON" : "OFF");
            elapsed_ms = 0;
        }

        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

/* =====================================================
 * app_main
 * ===================================================== */
void app_main(void)
{
    esp_err_t err = nvs_flash_init();
    if (err == ESP_ERR_NVS_NO_FREE_PAGES ||
        err == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        err = nvs_flash_init();
    }
    ESP_ERROR_CHECK(err);

    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());

    ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(IP_EVENT, IP_EVENT_STA_GOT_IP, &ip_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(MESH_EVENT, ESP_EVENT_ANY_ID, &mesh_event_handler, NULL));

    esp_netif_t *netif_sta = NULL;
    esp_netif_t *netif_ap = NULL;
    ESP_ERROR_CHECK(esp_netif_create_default_wifi_mesh_netifs(&netif_sta, &netif_ap));

    wifi_init_config_t wifi_cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&wifi_cfg));
    ESP_ERROR_CHECK(esp_wifi_set_storage(WIFI_STORAGE_FLASH));
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_APSTA));
    ESP_ERROR_CHECK(esp_wifi_set_ps(WIFI_PS_NONE));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_ERROR_CHECK(esp_mesh_init());

    mesh_cfg_t cfg = MESH_INIT_CONFIG_DEFAULT();
    uint8_t mid[6] = MESH_ID;
    memcpy(cfg.mesh_id.addr, mid, 6);

    cfg.channel = ROUTER_CH;
    cfg.router.ssid_len = strlen(ROUTER_SSID);
    memcpy(cfg.router.ssid, ROUTER_SSID, cfg.router.ssid_len);
    memcpy(cfg.router.password, ROUTER_PASS, strlen(ROUTER_PASS));

    cfg.mesh_ap.max_connection = MESH_MAX_CONN;
    cfg.mesh_ap.nonmesh_max_connection = 0;
    memcpy(cfg.mesh_ap.password, MESH_AP_PASS, strlen(MESH_AP_PASS));

    ESP_ERROR_CHECK(esp_mesh_set_config(&cfg));
    ESP_ERROR_CHECK(esp_mesh_set_self_organized(true, false));
    ESP_ERROR_CHECK(esp_mesh_start());

    hcsr04_init();
    dht11_init();
    yf_s201_init();

    xTaskCreate(hcsr04_task, "hcsr04_task", 4096, NULL, 5, NULL);
    xTaskCreate(dht11_task, "dht11_task", 4096, NULL, 5, NULL);
    xTaskCreate(yf_s201_task, "yf201_task", 4096, NULL, 5, NULL);

    ESP_LOGI(TAG, "Nodo HC-SR04 + DHT11 + YF-S201 compatible con root actual");
    ESP_LOGI(TAG, "HC-SR04(19/18) DHT11(17) YF-S201(16)");
    ESP_LOGI(TAG, "sizeof(msg_t)=%u | SEND_PERIOD_MS=%d",
             (unsigned)sizeof(msg_t), SEND_PERIOD_MS);
}
