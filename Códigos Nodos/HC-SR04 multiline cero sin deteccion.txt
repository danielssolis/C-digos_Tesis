// main.c - NODO HIJO (HC-SR04) para ESP-MESH (NO ROOT) + envío FLOAT al ROOT cada 10s
// ESP-IDF v6.1
// ✅ Envío constante cada SEND_PERIOD_MS
// ✅ Protocolo compatible (msg_t packed, value_type float)
// ✅ HC-SR04: envía distancia en cm (float)
// ✅ Envía 0 cuando no hay lectura correcta

#include <string.h>
#include <inttypes.h>
#include <stdbool.h>
#include <time.h>

#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#include "esp_log.h"
#include "esp_event.h"
#include "esp_wifi.h"
#include "esp_mesh.h"
#include "esp_netif.h"
#include "nvs_flash.h"
#include "esp_timer.h"
#include "esp_rom_sys.h"
#include "esp_err.h"
#include "esp_random.h"

#include "driver/gpio.h"
#include <math.h>

static const char *TAG = "HCSR04_Sala";

/* =========================
 *  MESH CONFIG
 * ========================= */
#define MESH_ID        {0x32,0x10,0x88,0x44,0x55,0x66}
#define ROUTER_SSID    "Pdegrado24"
#define ROUTER_PASS    "20Pgrado26$"
#define ROUTER_CH      0
#define MESH_AP_PASS   "v2345678"
#define MESH_MAX_CONN  6

/* =========================
 *  SENSOR PROTOCOL
 * ========================= */
typedef struct __attribute__((packed)) {
    uint8_t value_type;
    char node_name[24];
    char zone[24];
    char sensor[24];
    char metric[16];
    char unit[16];
    union {
        int32_t i;
        float   f;
    } v;
    uint32_t timestamp;
} msg_t;

/* =========================
 *  HC-SR04 CONFIG
 * ========================= */
#define TRIG 27
#define ECHO 26
#define SEND_PERIOD_MS 10000
#define ECHO_WAIT_TIMEOUT_US 30000
#define MAX_DISTANCE_CM 400.0f

/* =========================
 *  NODE IDENTITY
 * ========================= */
#define NODE_NAME   "Sala_nodo_3"
#define NODE_ZONE   "Sala"
#define SENSOR_NAME "hc_sr04_Sala"

/* =====================================================
 *  HELPERS
 * ===================================================== */
static void safe_copy(char *dst, size_t sz, const char *src)
{
    memset(dst, 0, sz);
    if (src) snprintf(dst, sz, "%s", src);
}

static uint32_t get_ts()
{
    time_t now = time(NULL);
    if ((uint32_t)now > 1000000000UL) return now;
    return esp_timer_get_time() / 1000000ULL;
}

/* =====================================================
 *  MESH EVENTS
 * ===================================================== */
static void wifi_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != WIFI_EVENT) return;

    if (id == WIFI_EVENT_STA_DISCONNECTED) {
        ESP_LOGW(TAG, "WiFi desconectado");
    }
}

static void ip_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (id == IP_EVENT_STA_GOT_IP) {
        ip_event_got_ip_t *e = (ip_event_got_ip_t *)data;
        ESP_LOGI(TAG, "IP: " IPSTR, IP2STR(&e->ip_info.ip));
    }
}

static void mesh_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    switch (id) {
    case MESH_EVENT_STARTED:
        ESP_LOGI(TAG, "Mesh iniciado");
        break;

    case MESH_EVENT_PARENT_CONNECTED:
        ESP_LOGI(TAG, "Parent conectado. layer=%d", esp_mesh_get_layer());
        break;

    case MESH_EVENT_PARENT_DISCONNECTED:
        ESP_LOGW(TAG, "Parent desconectado. layer=%d", esp_mesh_get_layer());
        break;

    default:
        break;
    }
}

/* =====================================================
 *  MESH SEND FLOAT
 * ===================================================== */
static void mesh_send_distance(float distance)
{
    if (esp_mesh_get_layer() <= 0) {
        ESP_LOGW(TAG, "No mesh layer, no envío");
        return;
    }

    msg_t msg;
    memset(&msg, 0, sizeof(msg));

    msg.value_type = 1; // FLOAT

    safe_copy(msg.node_name, sizeof(msg.node_name), NODE_NAME);
    safe_copy(msg.zone, sizeof(msg.zone), NODE_ZONE);
    safe_copy(msg.sensor, sizeof(msg.sensor), SENSOR_NAME);
    safe_copy(msg.metric, sizeof(msg.metric), "distance");
    safe_copy(msg.unit, sizeof(msg.unit), "cm");

    msg.v.f = distance;
    msg.timestamp = get_ts();

    mesh_data_t data = {
        .data = (uint8_t *)&msg,
        .size = sizeof(msg),
        .proto = MESH_PROTO_BIN,
        .tos = MESH_TOS_P2P
    };

    esp_err_t err = esp_mesh_send(NULL, &data, 0, NULL, 0);
    if (err == ESP_OK) {
        ESP_LOGI(TAG, "TX distance=%.2f cm", distance);
    } else {
        ESP_LOGW(TAG, "esp_mesh_send error: %s", esp_err_to_name(err));
    }
}

/* =====================================================
 *  HC-SR04 GPIO init
 * ===================================================== */
static void hcsr04_gpio_init(void)
{
    // TRIG salida
    gpio_config_t trig = {
        .pin_bit_mask = (1ULL << TRIG),
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    ESP_ERROR_CHECK(gpio_config(&trig));
    gpio_set_level(TRIG, 0);

    // ECHO entrada
    gpio_config_t echo = {
        .pin_bit_mask = (1ULL << ECHO),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    ESP_ERROR_CHECK(gpio_config(&echo));

    ESP_LOGI(TAG, "HC-SR04 listo TRIG=GPIO%d ECHO=GPIO%d", TRIG, ECHO);
}

/* =====================================================
 *  HC-SR04 medición distancia (cm)
 * ===================================================== */
static bool wait_for_level(gpio_num_t pin, int level, int64_t timeout_us)
{
    int64_t start = esp_timer_get_time();
    while (gpio_get_level(pin) != level) {
        if ((esp_timer_get_time() - start) > timeout_us) {
            return false;
        }
        esp_rom_delay_us(5);
    }
    return true;
}

static bool hcsr04_read_distance_cm(float *out_cm)
{
    // Pulso TRIG 10us
    gpio_set_level(TRIG, 0);
    esp_rom_delay_us(2);
    gpio_set_level(TRIG, 1);
    esp_rom_delay_us(10);
    gpio_set_level(TRIG, 0);

    // Esperar ECHO HIGH
    if (!wait_for_level(ECHO, 1, ECHO_WAIT_TIMEOUT_US)) {
        return false;
    }

    int64_t t_start = esp_timer_get_time();

    // Esperar ECHO LOW
    if (!wait_for_level(ECHO, 0, ECHO_WAIT_TIMEOUT_US)) {
        return false;
    }

    int64_t t_end = esp_timer_get_time();
    int64_t pulse_us = t_end - t_start;

    float cm = (float)pulse_us * 0.0343f * 0.5f;

    if (cm < 0.0f || cm > MAX_DISTANCE_CM) {
        return false;
    }

    *out_cm = cm;
    return true;
}

/* =====================================================
 *  TASK HC-SR04 -> MESH
 * ===================================================== */
static void hcsr04_task(void *arg)
{
    while (esp_mesh_get_layer() <= 0) {
        vTaskDelay(pdMS_TO_TICKS(200));
    }

    ESP_LOGI(TAG, "Nodo listo HC-SR04. Leyendo sensor cada %d ms...", SEND_PERIOD_MS);

    float last_send_value = -1.0f;  // Último valor enviado
    int64_t last_send_time = 0;     // Última vez que envió

    #define CHANGE_THRESHOLD 60.0f  // Detectar cambios >= 60cm

    while (1) {
        if (esp_mesh_get_layer() <= 0) {
            ESP_LOGW(TAG, "Sin mesh. Esperando reconexión...");
            while (esp_mesh_get_layer() <= 0) {
                vTaskDelay(pdMS_TO_TICKS(500));
            }
            ESP_LOGI(TAG, "Reconectado.");
        }

        float cm = 0.0f;
        bool ok = hcsr04_read_distance_cm(&cm);

        // Si falla lectura, envía 0 (sin detección)
        float send_val = ok ? cm : 0.0f;

        int64_t now = esp_timer_get_time() / 1000LL;  // Tiempo en ms
        bool should_send = false;
        const char *send_reason = "periódico";

        // Condición 1: Cambio vuelto a cero (de lectura válida a error)
        if (last_send_value > 0.0f && send_val == 0.0f) {
            should_send = true;
            send_reason = "transición a 0";
            ESP_LOGW(TAG, "Transición a 0 detectada: %.2f -> 0 cm", last_send_value);
        }

        // Condición 2: Cambio superior a 60cm
        if (last_send_value >= 0.0f && fabs(send_val - last_send_value) >= CHANGE_THRESHOLD) {
            should_send = true;
            send_reason = "cambio >= 60cm";
            ESP_LOGW(TAG, "Cambio grande detectado: %.2f -> %.2f cm (diff: %.2f)", 
                     last_send_value, send_val, fabs(send_val - last_send_value));
        }

        // Condición 3: Período de envío cumplido
        if ((now - last_send_time) >= SEND_PERIOD_MS) {
            should_send = true;
            send_reason = "periódico";
        }

        if (should_send) {
            mesh_send_distance(send_val);
            last_send_value = send_val;
            last_send_time = now;

            if (ok) {
                ESP_LOGI(TAG, "Distancia: %.2f cm [%s]", (double)cm, send_reason);
            } else {
                ESP_LOGW(TAG, "Lectura HC-SR04 falló, enviado 0 [%s]", send_reason);
            }
        }

        vTaskDelay(pdMS_TO_TICKS(100));  // Poll más frecuente para detectar cambios rápidamente
    }
}

/* =====================================================
 *  app_main
 * ===================================================== */
void app_main(void)
{
    ESP_ERROR_CHECK(nvs_flash_init());
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());

    ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID, wifi_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(IP_EVENT, IP_EVENT_STA_GOT_IP, ip_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(MESH_EVENT, ESP_EVENT_ANY_ID, mesh_event_handler, NULL));

    /* WiFi init (APSTA requerido por mesh) */
    wifi_init_config_t wifi_cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&wifi_cfg));
    ESP_ERROR_CHECK(esp_wifi_set_storage(WIFI_STORAGE_FLASH));
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_APSTA));
    ESP_ERROR_CHECK(esp_wifi_set_ps(WIFI_PS_NONE));
    ESP_ERROR_CHECK(esp_wifi_start());

    /* Mesh init */
    ESP_ERROR_CHECK(esp_mesh_init());

    mesh_cfg_t mc = MESH_INIT_CONFIG_DEFAULT();
    uint8_t mid[6] = MESH_ID;
    memcpy(mc.mesh_id.addr, mid, 6);

    mc.channel = ROUTER_CH;

    mc.router.ssid_len = strlen(ROUTER_SSID);
    memcpy(mc.router.ssid, ROUTER_SSID, mc.router.ssid_len);
    memcpy(mc.router.password, ROUTER_PASS, strlen(ROUTER_PASS));

    mc.mesh_ap.max_connection = MESH_MAX_CONN;
    mc.mesh_ap.nonmesh_max_connection = 0;
    memcpy(mc.mesh_ap.password, MESH_AP_PASS, strlen(MESH_AP_PASS));

    ESP_ERROR_CHECK(esp_mesh_set_config(&mc));

    // Permitir que otros nodos se conecten pero NO promover a root
    ESP_ERROR_CHECK(esp_mesh_set_self_organized(true, false));
    ESP_ERROR_CHECK(esp_mesh_start());

    ESP_LOGI(TAG, "Nodo HC-SR04 iniciado. NO ROOT");

    hcsr04_gpio_init();

    xTaskCreate(hcsr04_task, "hcsr04_task", 4096, NULL, 5, NULL);

    ESP_LOGI(TAG, "HC-SR04 sistema listo");
}
