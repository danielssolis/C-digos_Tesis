static const char *TAG = "ACS712_Salatv";

/* ===== EVENT HANDLERS ===== */
static void wifi_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != WIFI_EVENT) return;

    switch (id) {
        case WIFI_EVENT_STA_START:
            ESP_LOGI(TAG, "WIFI_EVENT_STA_START");
            break;
        case WIFI_EVENT_STA_CONNECTED:
            ESP_LOGI(TAG, "WIFI_EVENT_STA_CONNECTED");
            break;
        case WIFI_EVENT_STA_DISCONNECTED:
            ESP_LOGW(TAG, "WIFI desconectado");
            break;
        default:
            break;
    }
}

static void mesh_event_handler(void *arg, esp_event_base_t base, int32_t id, void *data)
{
    if (base != MESH_EVENT) return;

    switch (id) {
        case MESH_EVENT_STARTED: {
            mesh_addr_t myid = {0};
            esp_mesh_get_id(&myid);
            ESP_LOGI(TAG, "🔗 Mesh iniciado ID=%02X:%02X:%02X:%02X:%02X:%02X",
                     myid.addr[0], myid.addr[1], myid.addr[2],
                     myid.addr[3], myid.addr[4], myid.addr[5]);
            break;
        }
        case MESH_EVENT_PARENT_CONNECTED:
            ESP_LOGI(TAG, "✅ Parent conectado. Layer=%d Root=%d",
                     esp_mesh_get_layer(), esp_mesh_is_root());
            break;
        case MESH_EVENT_PARENT_DISCONNECTED:
            ESP_LOGW(TAG, "⚠️  Parent desconectado. Layer=%d",
                     esp_mesh_get_layer());
            break;
        default:
            break;
    }
}

/* IDENTIDAD */
#define NODE_NAME   "sala_nodo_2"
#define NODE_ZONE   "Sala"
#define SENSOR_NAME "acs712_salatv"

/* MESH */
#define MESH_ID        {0x32,0x10,0x88,0x44,0x55,0x66}
#define ROUTER_SSID    "Pdegrado24"
#define ROUTER_PASS    "20Pgrado26$"
#define ROUTER_CH      0
#define MESH_AP_PASS   "v2345678"
#define MESH_MAX_CONN  6

/* SENSOR - ACS712 5A */
#define ADC_CH ADC_CHANNEL_6 // GPIO34
#define VPERAMP 0.185f  // ACS712 5A = 0.185 V/A
#define SEND_PERIOD_MS 1000
#define CURRENT_CHANGE_THRESHOLD 0.2f  // 200mA threshold

// Parámetros de calibración - RELAJADOS PARA 5A
#define ACS712_ADC_MIN_VALID   8
#define ACS712_ADC_MAX_VALID   4087
#define ACS712_MIN_VALID_RATIO 0.60f
#define ACS712_MAX_SAT_RATIO   0.20f
#define ACS712_MAX_DV_CLIP_V   1.20f
#define ACS712_OUTPUT_SMOOTH_A 0.10f    // Menos suavizado (era 0.35)
#define ACS712_ZERO_HOLD_A     0.02f    // MÁS BAJO (era 0.25)
#define ACS712_ZERO_RELEASE_A  0.05f    // MÁS BAJO (era 0.35)
#define ACS712_NOISE_TRACK_A   0.05f    // Más rápido (era 0.02)
#define ACS712_NOISE_FLOOR_A   0.01f    // MÁS BAJO (era 0.08)

/* PROTOCOLO ROOT */
typedef struct __attribute__((packed)) {
    uint8_t value_type;
    char node_name[24];
    char zone[24];
    char sensor[24];
    char metric[16];
    char unit[16];
    union {
        int32_t i;
        float f;
    } v;
    uint32_t timestamp;
} msg_t;

/* GLOBAL */
static adc_oneshot_unit_handle_t adc;
static float last_current = -1.0f;
static float offset_v = 1.65f;  // Offset calibrado
static float last_irms_smooth = 0.0f;  // Para suavizado exponencial

/* HELPERS */
static void copy(char *dst, int sz, const char *src)
{
    memset(dst, 0, sz);
    snprintf(dst, sz, "%s", src);
}

static uint32_t ts()
{
    time_t now = time(NULL);
    if ((uint32_t)now > 1000000000) return now;
    return esp_timer_get_time()/1000000;
}

/* UTILIDADES */
static inline bool is_raw_invalid(int raw)
{
    return (raw <= ACS712_ADC_MIN_VALID) || (raw >= ACS712_ADC_MAX_VALID);
}

static inline float raw_to_voltage(int raw)
{
    return (raw / 4095.0f) * 3.3f;
}

/* CALIBRACIÓN DEL OFFSET */
static void calibrate_offset()
{
    const int N = 2000;
    const int us_delay = 500;

    double sum_v = 0;
    int ok = 0;
    int sat = 0;
    int adc_raw = 0;

    ESP_LOGW(TAG, "📊 Calibrando OFFSET (sin carga, esperar...)");
    for (int i = 0; i < N; i++) {
        if (adc_oneshot_read(adc, ADC_CH, &adc_raw) == ESP_OK) {
            if (is_raw_invalid(adc_raw)) {
                sat++;
            } else {
                sum_v += raw_to_voltage(adc_raw);
                ok++;
            }
        }
        for (volatile int j = 0; j < us_delay; j++);
    }

    float valid_ratio = (float)ok / (float)N;
    if (ok <= 0 || valid_ratio < ACS712_MIN_VALID_RATIO) {
        ESP_LOGE(TAG, "Calibración FALLÓ: validas=%d/%d sat=%d. Usando %.3fV", ok, N, sat, offset_v);
        return;
    }

    float off = (float)(sum_v / ok);
    
    // Rango válido ampliado: 1.2V a 2.6V (para diferentes configuraciones ADC)
    // ACS712 5A típico está entre 1.5-1.7V pero puede variar
    if (off < 1.2f || off > 2.6f) {
        ESP_LOGE(TAG, "⚠️  Offset FUERA DE RANGO: %.3fV (esperado 1.2-2.6V). Mantiene %.3fV", off, offset_v);
        return;
    }

    offset_v = off;
    ESP_LOGI(TAG, "✅ Offset calibrado: %.3fV (validas=%d sat=%d)", offset_v, ok, sat);
}

/* LECTURA AC RMS */
static float read_current_rms()
{
    const int N = 2000;
    const int us_delay = 500;

    double sum_dv_sq = 0;
    double sum_v = 0;
    int ok = 0;
    int adc_raw = 0;
    int sat_low = 0;
    int sat_high = 0;
    int clipped = 0;

    // Lectura de muestras con rechazo de outliers
    for (int i = 0; i < N; i++) {
        if (adc_oneshot_read(adc, ADC_CH, &adc_raw) != ESP_OK) {
            for (volatile int j = 0; j < us_delay; j++);
            continue;
        }

        if (adc_raw <= ACS712_ADC_MIN_VALID) {
            sat_low++;
            for (volatile int j = 0; j < us_delay; j++);
            continue;
        }
        if (adc_raw >= ACS712_ADC_MAX_VALID) {
            sat_high++;
            for (volatile int j = 0; j < us_delay; j++);
            continue;
        }

        float v = raw_to_voltage(adc_raw);
        float dv = v - offset_v;
        
        // Rechazar si está demasiado lejos (clipping)
        if (fabsf(dv) > ACS712_MAX_DV_CLIP_V) {
            clipped++;
            for (volatile int j = 0; j < us_delay; j++);
            continue;
        }

        sum_v += v;
        sum_dv_sq += (double)dv * (double)dv;
        ok++;
        
        for (volatile int j = 0; j < us_delay; j++);
    }

    float sat_ratio = (float)(sat_low + sat_high) / (float)N;
    float valid_ratio = (float)ok / (float)N;

    // Validar ventana
    if (ok <= 0 || valid_ratio < ACS712_MIN_VALID_RATIO || sat_ratio > ACS712_MAX_SAT_RATIO) {
        ESP_LOGW(TAG, "⚠️  Ventana inválida (val=%d/%d sat=%d clip=%d)", ok, N, sat_low+sat_high, clipped);
        return 0.0f;
    }

    // Recalcular offset dinámicamente (MÁS CONSERVADOR)
    float mean_v = (float)(sum_v / ok);
    // Solo ajustar si está dentro del rango válido (1.2-2.6V)
    if (mean_v >= 1.2f && mean_v <= 2.6f) {
        offset_v = (0.99f * offset_v) + (0.01f * mean_v);  // Menos agresivo
    }

    // Calcular RMS
    float vrms = sqrtf((float)(sum_dv_sq / ok));
    float irms = vrms / VPERAMP;

    // Threshold: cualquier lectura menor a 150mA es CERO
    // (ruido del sensor + offset impreciso)
    if (irms < 0.15f) {
        irms = 0.0f;
    }

    // Suavizado mínimo para no perder datos reales
    float irms_smooth = (ACS712_OUTPUT_SMOOTH_A * irms) + ((1.0f - ACS712_OUTPUT_SMOOTH_A) * last_irms_smooth);
    last_irms_smooth = irms_smooth;

    return irms_smooth;
}

/* ENVIO */
static void send_current(float amps)
{
    if (esp_mesh_get_layer() <= 0) return;

    msg_t msg;
    memset(&msg, 0, sizeof(msg));

    msg.value_type = 1;

    copy(msg.node_name, sizeof(msg.node_name), NODE_NAME);
    copy(msg.zone, sizeof(msg.zone), NODE_ZONE);
    copy(msg.sensor, sizeof(msg.sensor), SENSOR_NAME);
    copy(msg.metric, sizeof(msg.metric), "current");
    copy(msg.unit, sizeof(msg.unit), "A");

    msg.v.f = amps;
    msg.timestamp = ts();

    mesh_data_t data = {
        .data = (uint8_t *)&msg,
        .size = sizeof(msg),
        .proto = MESH_PROTO_BIN,
        .tos = MESH_TOS_P2P
    };

    esp_mesh_send(NULL, &data, 0, NULL, 0);

    ESP_LOGI(TAG, "📤 TX: %.3fA (offset=%.3fV)", amps, offset_v);
}

/* TASK PRINCIPAL */
static void task(void *arg)
{
    ESP_LOGI(TAG, "Esperando conexion a malla...");
    
    // Esperar a estar conectado a la malla
    while (esp_mesh_get_layer() <= 0) {
        vTaskDelay(pdMS_TO_TICKS(500));
    }

    // Calibracion al conectar
    vTaskDelay(pdMS_TO_TICKS(1000));
    calibrate_offset();
    vTaskDelay(pdMS_TO_TICKS(500));

    ESP_LOGI(TAG, "✅ Nodo listo. Layer=%d", esp_mesh_get_layer());

    uint32_t last_send_time = 0;

    while (1) {
        // Verificar desconexion
        if (esp_mesh_get_layer() <= 0) {
            ESP_LOGW(TAG, "Desconectado de malla. Reconectando...");
            while (esp_mesh_get_layer() <= 0) {
                vTaskDelay(pdMS_TO_TICKS(500));
            }
            ESP_LOGI(TAG, "✅ Reconectado. Layer=%d", esp_mesh_get_layer());
            last_current = -1.0f;  // Reset para forzar envio inicial
        }

        float amps = read_current_rms();
        uint32_t current_time = esp_timer_get_time() / 1000;
        uint32_t time_since_last_send = current_time - last_send_time;

        bool should_send = false;
        const char *reason = "";

        // Primer envio
        if (last_current < 0) {
            should_send = true;
            reason = "INICIAL";
        }
        // Deteccion de cambio significativo (>= 200mA)
        else if (fabsf(amps - last_current) >= CURRENT_CHANGE_THRESHOLD) {
            should_send = true;
            reason = "CAMBIO";
            ESP_LOGI(TAG, "CAMBIO: %.3fA -> %.3fA (Delta=%.3fA)", 
                     last_current, amps, fabsf(amps - last_current));
        }
        // Envio periodico
        else if (time_since_last_send >= SEND_PERIOD_MS) {
            should_send = true;
            reason = "PERIODICO";
        }

        if (should_send) {
            send_current(amps);
            last_current = amps;
            last_send_time = current_time;
            ESP_LOGI(TAG, "TX: %.3fA (%s) [Layer: %d]", amps, reason, esp_mesh_get_layer());
        } else {
            ESP_LOGD(TAG, "%.3fA (sin envio)", amps);
        }

        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

/* MAIN */
void app_main(void)
{
    esp_err_t err = nvs_flash_init();
    if (err == ESP_ERR_NVS_NO_FREE_PAGES || err == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        err = nvs_flash_init();
    }
    ESP_ERROR_CHECK(err);

    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());

    // Registrar event handlers
    ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(MESH_EVENT, ESP_EVENT_ANY_ID, &mesh_event_handler, NULL));

    esp_netif_t *sta = NULL;
    esp_netif_t *ap = NULL;
    ESP_ERROR_CHECK(esp_netif_create_default_wifi_mesh_netifs(&sta, &ap));

    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));
    ESP_ERROR_CHECK(esp_wifi_set_storage(WIFI_STORAGE_FLASH));
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_APSTA));
    ESP_ERROR_CHECK(esp_wifi_set_ps(WIFI_PS_NONE));  // ← Desactiva power saving
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_ERROR_CHECK(esp_mesh_init());

    mesh_cfg_t mc = MESH_INIT_CONFIG_DEFAULT();

    uint8_t id[6] = MESH_ID;
    memcpy(mc.mesh_id.addr, id, 6);

    mc.channel = ROUTER_CH;

    mc.router.ssid_len = strlen(ROUTER_SSID);
    memcpy(mc.router.ssid, ROUTER_SSID, mc.router.ssid_len);
    memcpy(mc.router.password, ROUTER_PASS, strlen(ROUTER_PASS));

    mc.mesh_ap.max_connection = MESH_MAX_CONN;
    mc.mesh_ap.nonmesh_max_connection = 0;
    memcpy(mc.mesh_ap.password, MESH_AP_PASS, strlen(MESH_AP_PASS));

    ESP_ERROR_CHECK(esp_mesh_set_config(&mc));
    ESP_ERROR_CHECK(esp_mesh_set_self_organized(true, false));  // ← IMPORTANTE: false
    ESP_ERROR_CHECK(esp_mesh_start());
    
    ESP_LOGI(TAG, "🔗 Nodo inicializado - esperando conexión al ROOT");

    /* ADC */
    adc_oneshot_unit_init_cfg_t init = {
        .unit_id = ADC_UNIT_1
    };
    adc_oneshot_new_unit(&init, &adc);

    adc_oneshot_chan_cfg_t ch = {
        .bitwidth = ADC_BITWIDTH_12,
        .atten = ADC_ATTEN_DB_12
    };
    adc_oneshot_config_channel(adc, ADC_CH, &ch);

    // Calibración del offset del ADC (será hecha en el task)
    // Se hace en el task después de conectar a mesh

    xTaskCreate(task, "acs712", 4096, NULL, 5, NULL);

    ESP_LOGI(TAG, "ACS712 listo");
}
